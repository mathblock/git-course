----
git status = voir l'état des fichiers
----
git add . = ajouter tous les fichiers au prochain commit
----
git commit -m "Premier commit" = valider les changements
----
git checkout <hash_du_commit> = revenir à un commit précédent
----
git switch -c <nom de la nouvelle branche> = créer et se positionner sur une nouvelle branche
----
git switch <nom de la branche> = retourner sur une branche
----
// Avant de faire un merge, se positionner sur la branche cible
